Algorithm~\ref{alg:decision} states the access-decision procedure the main
article's Equation~(3) describes in full (moved here from the main text
to save space; two of its properties are summarised in the main article's
``Zone and temporal scoping'' section).

\begin{widealg}
\caption{Access decision (\texttt{CheckAccess})}
\label{alg:decision}
\small
\begin{algorithmic}[1]
\Require request $(u, \mathit{op}, \mathit{res}, t)$, caller certificate $C$
\Ensure \textsc{Grant} or \textsc{Deny}. Every path writes an audit entry,
  so every invocation mutates state and commits through ordering.
\State $n \gets \textproc{Nonce}(\mathit{req})$
\If{$\textproc{SeenBefore}(n)$}
  \State $\textproc{Audit}(\textsc{Deny}, \textsc{ReplayDetected})$;\;
         \Return \textsc{Deny}
\EndIf
\State $z_u \gets \textproc{ParseZoneASN1}(C)$
  \algcomment{DER bytes, not PEM text}
\If{$\textproc{OnCRL}(C)$}
  \State $\textproc{Audit}(\textsc{Deny}, \textsc{CertRevoked})$;\;
         \Return \textsc{Deny}
\EndIf
\State $A \gets \textproc{RoleAssignments}(u)$
\State $R_{\text{act}} \gets \emptyset$
\For{$a \in A$}
  \If{$a.\mathit{ExpiresAt} < t$}
    \State $\textproc{RevokeRole}(a)$
      \algcomment{state-mutating: forces endorsement}
  \Else
    \State $R_{\text{act}} \gets R_{\text{act}} \cup \{a.\mathit{role}\}$
  \EndIf
\EndFor
\If{$R_{\text{act}} = \emptyset$}
  \State $\textproc{Audit}(\textsc{Deny}, \textsc{RoleInsufficient})$;\;
         \Return \textsc{Deny}
\EndIf
\State $P \gets \bigcup_{r \in R_{\text{act}}}
        \bigcup_{r' \preceq r} pa(r')$
  \algcomment{inheritance closure, depth $\le 3$}
\If{$(\mathit{op}, \mathit{res}) \notin P$}
  \State $\textproc{Audit}(\textsc{Deny}, \textsc{OpNotPermitted})$;\;
         \Return \textsc{Deny}
\EndIf
\State $z_r \gets \textproc{ZoneOf}(\mathit{res})$
\If{$z_u \neq z_r \;\land\; \lnot\,\textproc{ValidCZT}(u, z_r, t)$}
  \State $\textproc{Audit}(\textsc{Deny}, \textsc{ZoneMismatch})$;\;
         \Return \textsc{Deny}
\EndIf
\State $\textproc{RecordNonce}(n)$;\;
       $\textproc{Audit}(\textsc{Grant}, (\mathit{op}, \mathit{res}, z_r))$
\State \Return \textsc{Grant}
\end{algorithmic}
\end{widealg}
